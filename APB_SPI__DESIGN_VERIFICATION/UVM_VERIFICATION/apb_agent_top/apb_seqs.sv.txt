//cphase =1 cpol =1 lsb =0

class msb_cpol1_cpha1_seqs extends apb_seqs;
        `uvm_object_utils(msb_cpol1_cpha1_seqs)
         bit [7:0]ctrl;

function new(string name="msb_cpol1_cpha1_seqs");
        super.new(name);
endfunction

task body();
         super.body();
            if(!uvm_config_db #(bit[7:0])::get(null,get_full_name(),"bit[7:0]",ctrl))
                `uvm_fatal(get_type_name(),"cannot get ctrl from uvm_config_db. Have you set it?")

        repeat(1)
             begin

                req=apb_xtn::type_id::create("req");

                start_item(req);
                assert(req.randomize() with {PRESETn==1'b1; PWRITE==1'b1; PADDR==3'b000; PWDATA==ctrl;})
                `uvm_info("APB_SEQ_GEN", $sformatf("Generated Transaction:\n%s", req.sprint()), UVM_LOW)
                finish_item(req);


                start_item(req);
                assert(req.randomize() with {PRESETn==1'b1; PWRITE==1'b1; PADDR==3'b001; PWDATA==8'b0001_1011;})
                `uvm_info("APB_SEQ_GEN", $sformatf("Generated Transaction:\n%s", req.sprint()), UVM_LOW)
                finish_item(req);


                start_item(req);
                assert(req.randomize() with {PRESETn==1'b1; PWRITE==1'b1; PADDR==3'b010; PWDATA==8'b0001_0010;})
                `uvm_info("APB_SEQ_GEN", $sformatf("Generated Transaction:\n%s", req.sprint()), UVM_LOW)
                finish_item(req);

                start_item(req);
                assert(req.randomize() with {PRESETn==1'b1; PWRITE==1'b0; PADDR==3'b010;})
                `uvm_info("APB_SEQ_GEN", $sformatf("Generated Transaction:\n%s", req.sprint()), UVM_LOW)
                finish_item(req);


                start_item(req);
                assert(req.randomize() with {PRESETn==1'b1; PWRITE==1'b1; PADDR==3'b101; PWDATA inside {[8'h00:8'h0f]};})
                `uvm_info("APB_SEQ_GEN", $sformatf("Generated Transaction:\n%s", req.sprint()), UVM_LOW)
                finish_item(req);


        end
endtask
endclass
