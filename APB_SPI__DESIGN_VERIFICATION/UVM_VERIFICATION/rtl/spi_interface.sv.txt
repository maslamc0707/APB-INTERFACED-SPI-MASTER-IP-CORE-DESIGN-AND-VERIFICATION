interface spi_intf(input bit clock);
   logic ss;
   logic sclk;
   logic mosi;
   logic miso;
//   logic spi_interrupt_request;

 //clocking block for spi driver

   clocking spi_drv_cb@(posedge clock);
     default input #1 output #1;
     input ss;
     input sclk;
     input mosi;
     output miso;
  //   input spi_interrupt_request;
   endclocking

 //clocking block for spi monitor

   clocking spi_mon_cb@(posedge clock);
     default input #1 output #1;
     input ss;
     input sclk;
     input mosi;
     input miso;
    // input spi_interrupt_request;
   endclocking

 //modport for spi

   modport SPI_DRV_MP (clocking spi_drv_cb);
   modport SPI_MON_MP (clocking spi_mon_cb);
 endinterface
