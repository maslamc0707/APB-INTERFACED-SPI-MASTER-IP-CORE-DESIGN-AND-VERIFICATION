class spi_monitor extends uvm_monitor;
        `uvm_component_utils(spi_monitor)

        spi_agt_config a_cfg;
        virtual spi_intf.SPI_MON_MP vif;
        uvm_analysis_port#(spi_xtn) monitor_port;

        bit [7:0]ctrl;
        bit cphase;
        bit cpol;
        bit lsb;

function new(string name="spi_monitor",uvm_component parent);
        super.new(name,parent);
        monitor_port=new("monitor_port",this);

endfunction : new

function void build_phase(uvm_phase phase);
        super.build_phase(phase);

        if(!uvm_config_db #(spi_agt_config)::get(this,"","spi_agt_config",a_cfg))
        `uvm_fatal(get_type_name(),"monitor not getting")

endfunction


function void connect_phase(uvm_phase phase);
        vif=a_cfg.vif;
endfunction : connect_phase

task run_phase(uvm_phase phase);
        forever

              collect_data();

endtask


task collect_data();

 spi_xtn xtn;
 int bit_idx;

xtn = spi_xtn::type_id::create("xtn");

if(!uvm_config_db #(bit[7:0])::get(this,"","bit[7:0]", ctrl))
        `uvm_fatal(get_type_name(),"cannot getting ctrn")

        cphase=ctrl[2];
        cpol=ctrl[3];
        lsb=ctrl[0];

`uvm_info(get_type_name(),$sformatf("SPI MONITOR CTRL=%0d",ctrl),UVM_LOW)

// wait for slave select

wait(!vif.spi_mon_cb.ss);


//collectr all 8 bits

for (int i = 0; i < 8; i++) begin

bit_idx = lsb ? i : (7-i);

        if(cphase ^ cpol)
          @(negedge vif.spi_mon_cb.sclk);
        else
          @(posedge vif.spi_mon_cb.sclk);

                xtn.miso[bit_idx] = vif.spi_mon_cb.miso;
                xtn.mosi[bit_idx] = vif.spi_mon_cb.mosi;
                xtn.ss = vif.spi_mon_cb.ss;
end

`uvm_info("SPI_MONITOR",$sformatf("transaction received from spi slave \n %s",xtn.sprint()),UVM_LOW)

$display("SPI_MONITOR : MOSI=%0h MISO=%0h", xtn.mosi, xtn.miso);

        monitor_port.write(xtn);
        a_cfg.mon_rcvd_xtn_cnt++;

        @(vif.spi_mon_cb);
endtask

function void report_phase(uvm_phase phase);
        `uvm_info(get_type_name(),$sformatf("SPI MONITOR:no of transaction received in spi monitor: %d",a_cfg.mon_rcvd_xtn_cnt++),UVM_LOW)
endfunction
endclass
