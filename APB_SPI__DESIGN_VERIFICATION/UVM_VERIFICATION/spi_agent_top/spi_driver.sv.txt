class spi_driver extends uvm_driver#(spi_xtn);
        `uvm_component_utils(spi_driver)

        spi_agt_config a_cfg;
        virtual spi_intf.SPI_DRV_MP vif;

        bit [7:0]ctrl;
        bit cphase;
        bit cpol;
        bit lsb;

function new(string name="spi_driver",uvm_component parent);
        super.new(name,parent);
endfunction : new

function void build_phase(uvm_phase phase);
        super.build_phase(phase);
        if(!uvm_config_db#(spi_agt_config)::get(this,"","spi_agt_config",a_cfg))
        `uvm_fatal(get_type_name(),"not getting");

endfunction : build_phase

function void connect_phase(uvm_phase phase);
        vif=a_cfg.vif;
endfunction : connect_phase


task run_phase(uvm_phase phase);
        forever
         begin
$display("before get next time");


        seq_item_port.get_next_item(req);

$display("after get next item");


        send_to_dut(req);
       `uvm_info("SPI_WR_DRIVER","printing from driver ",UVM_LOW)
        req.print();
        seq_item_port.item_done();
        end
endtask : run_phase

task send_to_dut(spi_xtn xtn);

        if(!uvm_config_db#(bit[7:0])::get(this,"","bit[7:0]", ctrl))
        `uvm_fatal(get_type_name(),"spi slave driver")

        cphase=ctrl[2];
        cpol=ctrl[3];
        lsb=ctrl[0];

// wait for slave select

`uvm_info("DRV_DEBUG", $sformatf("Waiting for SS to go low. Current SS = %0b", vif.spi_drv_cb.ss), UVM_LOW)

    wait(!vif.spi_drv_cb.ss==1)

`uvm_info("DRV_DEBUG", "SS is now LOW. Waiting for SCLK edges...", UVM_LOW)

//drive 1st bit immediately when cphase = 0

if (!cphase)
     vif.spi_drv_cb.miso <= xtn.miso[lsb ? 0 : 7];

//driev all 8 bit

for (int i=(cphase?0:1); i < 8; i++) begin

$display("i=%0d CPOL=%0d CPHA=%0d Waiting Edge Time=%0t",
            i, cpol, cphase, $time);

        if (cphase ^ cpol)
                @(posedge vif.spi_drv_cb.sclk);
        else
                @(negedge vif.spi_drv_cb.sclk);

$display("Edge Detected i=%0d SCLK=%0b Time=%0t",
            i, vif.spi_drv_cb.sclk, $time);

vif.spi_drv_cb.miso <= xtn.miso[lsb ? i : (7-i)];
end

a_cfg.drv_data_send_cnt++;

endtask

endclass
